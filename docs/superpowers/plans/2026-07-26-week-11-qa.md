## Task 5: RLS audit

- [ ] **`tests/rls/helpers.ts`**

```ts
import { createClient } from '@supabase/supabase-js';

const url = process.env.PUBLIC_SUPABASE_URL!;
const anonKey = process.env.PUBLIC_SUPABASE_ANON_KEY!;
const adminClient = createClient(url, process.env.SUPABASE_SERVICE_ROLE_KEY!);

export function anonClient() {
  return createClient(url, anonKey);
}
export async function signedInAs(email: string, password: string) {
  const c = createClient(url, anonKey, {
    auth: { persistSession: false, autoRefreshToken: false },
  });
  const { error } = await c.auth.signInWithPassword({ email, password });
  if (error) throw error;
  return c;
}
export { adminClient };
```

- [ ] **One test file per major table**. Example `tests/rls/orders.rls.test.ts`:

```ts
import { describe, it, expect, beforeAll, afterAll } from 'vitest';
import { makeUser, dropUser } from '../fixtures/users';
import { signedInAs, anonClient } from './helpers';

let buyer: any, otherBuyer: any, ops: any;
beforeAll(async () => {
  buyer = await makeUser('customer');
  otherBuyer = await makeUser('customer');
  ops = await makeUser('operations');
});
afterAll(async () => {
  await Promise.all([dropUser(buyer.id), dropUser(otherBuyer.id), dropUser(ops.id)]);
});

describe('orders RLS', () => {
  it('anonymous cannot SELECT orders', async () => {
    const r = await anonClient().from('orders').select('id').limit(1);
    expect(r.data).toEqual([]); // RLS hides all
  });
  it('customer cannot see orders of another customer', async () => {
    // Seed an order for otherBuyer via service-role
    // ... insert order with profile_id = otherBuyer.id ...
    const c = await signedInAs(buyer.email, buyer.password);
    const r = await c.from('orders').select('id').limit(10);
    // None should belong to otherBuyer
  });
  it('operations sees all orders', async () => {
    const c = await signedInAs(ops.email, ops.password);
    const r = await c.from('orders').select('id').limit(10);
    expect((r.data ?? []).length).toBeGreaterThan(0);
  });
});
```
